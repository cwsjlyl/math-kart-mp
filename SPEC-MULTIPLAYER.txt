四、嵌入      index.html 的連線模組程式碼架構（PeerJS 範例）
只需在現有   index.html 引入輕量級 PeerJS CDN，即可加入以下 NetworkManager 模組與現有 allKarts 無縫接合：

<!-- 1. 在 <head> 加入 PeerJS 輕量連線庫 -->
<script src="https://unpkg.com/peerjs@1.5.4/dist/peerjs.min.js"></script>
// 2. 多人連線管理模組（支援 4 位數房間碼、20Hz 賽車同步、道具與香蕉事件廣播）
class KartNetworkManager {
  constructor() {
    this.peer = null;
    this.isHost = false;
    this.roomCode = '';
    this.mySlot = 0; // 0: 房主, 1~3: 加入的同學
    this.connections = new Map(); // peerId -> DataConnection
    this.hostConn = null;
    this.remoteStates = new Map(); // slot -> 最新網路封包
    this.syncAccumulator = 0;
  }

 // 房主建立 4 位數房間（例如 5821 -> Peer ID: cwsj-mk8-5821）
 createRoom(playerProfile, onRoomReady, onLobbyUpdate) {
   this.isHost = true;
   this.mySlot = 0;
   this.roomCode = String(Math.floor(1000 + Math.random() * 9000));
   const peerId = `cwsj-mk8-${this.roomCode}`;

     this.peer = new Peer(peerId);
     this.peer.on('open', () => onRoomReady(this.roomCode));

     this.peer.on('connection', (conn) => {
       if (this.connections.size >= 3) {
         conn.on('open', () => conn.send({ type: 'ROOM_FULL' }));
         return;
       }
       const assignedSlot = this.connections.size + 1;
       conn.slot = assignedSlot;
       this.connections.set(conn.peer, conn);

       conn.on('data', (msg) => this.handleHostMessage(conn, msg, onLobbyUpdate));
       conn.on('close', () => {
         this.connections.delete(conn.peer);
         onLobbyUpdate(this.getLobbyRoster());
       });
     });
 }

 // 學生輸入 4 位數房間碼加入
 joinRoom(code, playerProfile, onJoined, onMessage) {
   this.isHost = false;
   this.roomCode = code.trim();
   this.peer = new Peer();

     this.peer.on('open', () => {
       const conn = this.peer.connect(`cwsj-mk8-${this.roomCode}`, { reliable: false });
       this.hostConn = conn;
       conn.on('open', () => {
         conn.send({ type: 'JOIN_REQ', profile: playerProfile });
       });
       conn.on('data', (msg) => {
         if (msg.type === 'WELCOME') {
           this.mySlot = msg.slot;
           onJoined(msg);
         } else {
           onMessage(msg);
         }
      });
    });
}

// 每幀呼叫：控制以 20Hz（每 0.05 秒）發送本地賽車座標狀態
tickSync(dt, localKart) {
  this.syncAccumulator += dt;
  if (this.syncAccumulator < 0.05) return;
  this.syncAccumulator = 0;

    const statePacket = {
      type: 'KART_STATE',
      slot: this.mySlot,
      p: Number(localKart.totalProgress.toFixed(5)),
      o: Number(localKart.offset.toFixed(2)),
      v: Number(localKart.speed.toFixed(1)),
      s: Number(localKart.smoothSteer.toFixed(2)),
      boost: localKart.boostTimer > 0 ? 1 : 0,
      star: localKart.starTimer > 0 ? 1 : 0,
      gold: localKart.goldenTimer > 0 ? 1 : 0,
      shrink: localKart.shrinkTimer > 0 ? 1 : 0,
      slow: localKart.slowTimer > 0 ? 1 : 0,
      item: localKart.heldItem || ''
    };

    this.broadcastOrSendToHost(statePacket);
}

// 廣播道具事件（拋出香蕉皮、發射紅藍綠龜殼、閃電、墨汁）
sendItemEvent(eventPayload) {
  this.broadcastOrSendToHost({
    type: 'ITEM_EVENT',
    senderSlot: this.mySlot,
    ...eventPayload
  });
}

broadcastOrSendToHost(packet) {
  if (this.isHost) {
    for (const conn of this.connections.values()) {
      if (conn.open) conn.send(packet);
    }
  } else if (this.hostConn && this.hostConn.open) {
    this.hostConn.send(packet);
  }
}

handleHostMessage(conn, msg, onLobbyUpdate) {
  if (msg.type === 'JOIN_REQ') {
    conn.profile = msg.profile;
    conn.send({ type: 'WELCOME', slot: conn.slot, trackId: selectedTrackId, mathRange: mathRangeMode });
    const roster = this.getLobbyRoster();
    this.broadcastOrSendToHost({ type: 'LOBBY_ROSTER', roster });
    if (onLobbyUpdate) onLobbyUpdate(roster);
    return;
  }
  // 房主收到某位學生的賽車座標或道具事件後，立即轉發給房間內其他同學
  for (const otherConn of this.connections.values()) {
    if (otherConn !== conn && otherConn.open) otherConn.send(msg);
        }
        this.applyIncomingPacket(msg);
    }

    applyIncomingPacket(msg) {
      if (msg.type === 'KART_STATE') {
        this.remoteStates.set(msg.slot, { ...msg, recvTime: performance.now() });
      }
    }

    getLobbyRoster() {
      const list = [{ slot: 0, name: '房主 (P1)' }];
      for (const conn of this.connections.values()) {
        list.push({ slot: conn.slot, ...(conn.profile || { name: `同學 P${conn.slot + 1}` }) });
      }
      return list;
    }
}


五、      iPad 課堂實戰三大避坑要點
    1. 防止房主切換分頁導致休眠（Background Throttling）：
          在 iPad Safari 上，若擔任「房主（Host ）」的學生按了 Home 鍵或切換分頁，瀏覽器會暫停 requestAnimationFrame，導致

          其他同學收不到 AI 與轉發封包。

             解法：房主的網絡轉發邏輯綁定在             PeerJS 的 conn.on('data') 事件回呼與 Web Worker 計時器上，即使畫面短暫縮小仍能持
             續轉發封包；並使用Wake Lock API（navigator.wakeLock.request('screen')）防止比賽中途螢幕自動變暗鎖機。
    2. 名字輸入與車頂名牌（Name Tag Sprite）：
          進入房間前讓學生輸入簡短姓名或學號（例如 4A 陳小明），利用 Three.js CanvasTexture + Sprite 懸浮顯示在每輛賽車頭

          頂上方（與現有 v4.0 的 AI 道具氣泡並列），比賽時一眼認出旁邊擦撞自己的是哪位同學！

    3. 單機與連線雙模式並存：
          保留原本的「單人挑戰 AI」按鈕，當學生在家溫習或沒有同學在線時，依然能隨時單人開跑並進行賽後 4 倍錯題特訓。
